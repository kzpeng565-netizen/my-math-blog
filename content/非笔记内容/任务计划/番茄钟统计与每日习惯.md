const pages = dv.pages();
const emoji = "🍅";

// 获取当前日期和上周同一天（8天前）
const today = new Date();
const lastWeekSameDay = new Date();
lastWeekSameDay.setDate(today.getDate() - 7); // 包含今天，所以是7天前

// 格式化日期为YYYY-MM-DD
const formatDate = (date) => date.toISOString().split('T')[0];

const todayStr = formatDate(today);
const lastWeekSameDayStr = formatDate(lastWeekSameDay);

// 星期几缩写的映射
const dayAbbreviations = {
  0: "Sun",
  1: "Mon", 
  2: "Tue",
  3: "Wed",
  4: "Thu",
  5: "Fri",
  6: "Sat"
};

// 获取日期对应的星期几缩写
const getDayOfWeek = (dateStr) => {
  const date = new Date(dateStr);
  return dayAbbreviations[date.getDay()];
};

// 将分钟数转换为"X小时Y分钟"格式
const formatMinutesToHours = (minutes) => {
  const hours = Math.floor(minutes / 60);
  const mins = minutes % 60;
  
  if (hours > 0 && mins > 0) {
    return `${hours}小时${mins}分钟`;
  } else if (hours > 0) {
    return `${hours}小时`;
  } else {
    return `${mins}分钟`;
  }
};

dv.table(
  ["Date", "Pomodoros", "Total"],
  pages.file.lists
    .filter((item) => item?.pomodoro == "WORK")
    .filter((item) => {
      if (item.end && item.end.length >= 10) {
        const dateStr = item.end.substring(0, 10);
        // 只保留最近8天（包含今天和上周同一天）的数据
        return dateStr >= lastWeekSameDayStr && dateStr <= todayStr;
      }
      return false;
    })
    .groupBy((item) => item.end.substring(0, 10))
    .sort((group) => group.key, 'desc') // 按日期降序排列，最新的在前
    .map((group) => {
      let sum = 0;
      group.rows.forEach((row) => (sum += row.duration.as("minutes")));
      const totalMinutes = Math.round(sum);
      return [
        `${group.key} (${getDayOfWeek(group.key)})`, // 添加星期几缩写
        group.rows.length > 5
          ? `${emoji}  ${group.rows.length}`
          : `${emoji.repeat(group.rows.length)}`,
        formatMinutesToHours(totalMinutes),
      ];
    })
)
